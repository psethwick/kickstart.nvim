local source = {}
local comrade = require('comrade')
local async = require('plenary.async')

source.new = function()
  return setmetatable({}, { __index = source })
end

-- Mark this source as async since we'll be waiting for IDE responses
source.is_async = true

function source:get_debug_name()
  return 'comrade_completion'
end

function source:get_keyword_pattern()
  -- Match any word characters
  return [[\k\+]]
end

function source:is_available()
  -- Check if we're in a Comrade-managed buffer
  local bufnr = vim.api.nvim_get_current_buf()
  return comrade.is_comrade_buffer(bufnr)
end

-- Convert JetBrains completion item to nvim-cmp format
local function convert_completion_item(item)
  return {
    label = item.label,
    kind = item.kind or vim.lsp.protocol.CompletionItemKind.Text,
    documentation = {
      kind = 'markdown',
      value = item.documentation or ''
    },
    insertText = item.insertText or item.label,
    data = item.data
  }
end

function source:complete(params, callback)
  -- If we're not in a valid position, return empty results
  if not self:is_available() then
    callback({ items = {} })
    return
  }

  local bufnr = params.context.bufnr
  local pos = params.context.cursor_position

  -- Request completions from Comrade
  async.run(function()
    local items = {}
    
    -- Get completion items from Comrade
    local completions = comrade.request_completions(bufnr, {
      line = pos[1] - 1,
      character = pos[2] - 1
    })

    if completions then
      -- Convert each completion item to nvim-cmp format
      for _, item in ipairs(completions) do
        table.insert(items, convert_completion_item(item))
      end
    end

    callback({ items = items })
  end)
end

-- Handle completion confirmation
function source:execute(completion_item, callback)
  if completion_item.data then
    comrade.apply_completion(completion_item.data)
  end
  callback(completion_item)
end

-- Register the source
require('cmp').register_source('comrade', source.new())

-- Example configuration
--[[
require('cmp').setup({
  sources = {
    { name = 'comrade', priority = 1000 }, -- High priority for IDE completions
    -- other sources...
  },
})
--]]
